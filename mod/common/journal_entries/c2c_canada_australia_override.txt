REPLACE:je_canada_can = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		hidden_trigger = {
			NOT = { c:NEW ?= THIS }
		}
		OR = {
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:franco_canadian
			country_has_primary_culture = cu:acadian
		}
		is_subject = yes
	}

	is_shown_in_lobby = {
		hidden_trigger = {
			NOT = { c:NEW ?= THIS }
		}
		OR = {
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:franco_canadian
			country_has_primary_culture = cu:acadian
		}
	}

	possible = {
		has_technology_researched = nationalism
		hidden_trigger = {
			NOT = { c:NEW ?= THIS }
		}
	}

	scripted_button = je_canada_unite_can_button

	complete = {
		owns_entire_state_region = STATE_ONTARIO
		owns_entire_state_region = STATE_ONTARIO_PENINSULA
		owns_entire_state_region = STATE_ALGOMA
		owns_entire_state_region = STATE_NORTHWEST_ONTARIO
		owns_entire_state_region = STATE_NEW_BRUNSWICK
		owns_entire_state_region = STATE_NOVA_SCOTIA
		owns_entire_state_region = STATE_QUEBEC
		owns_entire_state_region = STATE_SAINT_LAURENT
		owns_entire_state_region = STATE_QUEBEC_SAGUENAY
		owns_entire_state_region = STATE_NOUVEAU_QUEBEC
		owns_entire_state_region = STATE_ALBERTA
		owns_entire_state_region = STATE_EDMONTON
		owns_entire_state_region = STATE_BRITISH_COLUMBIA
		owns_entire_state_region = STATE_BC_INTERIOR
		owns_entire_state_region = STATE_VANCOUVER_ISLAND
		owns_entire_state_region = STATE_SASKATCHEWAN
		owns_entire_state_region = STATE_MANITOBA
	}

	on_complete = {
		trigger_event = { id = can_aus.5 }
		change_tag = CAN
		add_primary_culture = cu:anglo_canadian
		add_primary_culture = cu:franco_canadian
		add_primary_culture = cu:acadian
		hidden_effect = {
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_ONTARIO
					}
				}
			}
			s:STATE_ONTARIO = {
				random_scope_state = {
			        set_hub_name = {
			            type = city
			            name = HUB_NAME_STATE_ONTARIO_city_ottawa
			        }
				}
			}
		}
	}

	invalid = {
		OR = {
			is_subject = no
			hidden_trigger = {
				c:NEW ?= THIS
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}
