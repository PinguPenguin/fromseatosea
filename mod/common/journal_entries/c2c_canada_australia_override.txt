REPLACE:je_canada_can = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_historical_content
	should_be_pinned_by_default = yes
	should_be_pinned_by_default_uninvolved_or_context = yes

	progressbar = yes

	scripted_button = je_c2c_confederation_button_conference
	scripted_button = je_c2c_confederation_button_confederation

	is_shown_when_inactive = {
		c2c_is_confederation_candidate = yes
	}

	is_shown_in_lobby = {
		is_subject = yes
		capital = {
			is_in_geographic_region = geographic_region_greater_canada
		}
		OR = {
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:franco_canadian
			country_has_primary_culture = cu:acadian
		}
		hidden_trigger = {
			NOT = { c:NEW ?= THIS }
			NOT = { c:CAN ?= THIS }
		}
	}

	possible = {
		has_technology_researched = nationalism
		custom_tooltip = {
			text = c2c_confederation_candidate_count_tt
			any_country = {
				c2c_is_confederation_candidate = yes
				count > 2
			}
		}
	}

	immediate = {
		c2c_confederation_initialize_interest = yes
	}

	complete = {
		custom_tooltip = {
			text = c2c_confederation_progress_complete_tt
			global_var:c2c_confederation_interest >= 100
		}
		custom_tooltip = {
			text = c2c_confederation_all_colonies_absorbed_tt
			NOT = {
				any_country = {
					c2c_is_confederation_partner = yes
					NOT = { THIS = ROOT }
				}
			}
		}
	}

	on_complete = {
		set_variable = {
			name = c2c_confederation_initiator
			days = 365
		}
		trigger_event = { id = c2c_confederation.3 popup = yes }
	}

	on_monthly_pulse = {
		effect = {
			c2c_confederation_add_interest_for_scope = yes
		}
	}

	current_value = {
		add = global_var:c2c_confederation_interest
	}

	goal_add_value = {
		value = 100
	}

	invalid = {
		OR = {
			is_subject = no
			hidden_trigger = {
				OR = {
					c:NEW ?= THIS
					c:CAN ?= THIS
				}
			}
		}
	}

	weight = 100
}
