# Confederation: c2c's version of vanilla's je_canada.
REPLACE:je_canada = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_global_international_situations

	activation_description = {
		always = yes
	}

	should_be_involved = {
		hidden_trigger = {
			NOT = { c:NEW ?= THIS }
		}
		OR = {
			# The Canadian colonies
			AND = {
				capital = {
					is_in_geographic_region = geographic_region_greater_canada
				}
				is_subject = yes
				country_is_colonial_or_company = yes
			}
			# Their overlord
			AND = {
				NOT = {
					capital = {
						is_in_geographic_region = geographic_region_greater_canada
					}
				}
				any_subject_or_below = {
					hidden_trigger = {
						NOT = { c:NEW ?= THIS }
					}
					capital = {
						is_in_geographic_region = geographic_region_greater_canada
					}
					country_is_colonial_or_company = yes
					count >= 2
				}
			}
		}
	}

	should_show_when_not_involved = {
		hidden_trigger = {
			NOT = { c:NEW ?= THIS }
		}
		OR = {
			AND = {
				capital = {
					is_in_geographic_region = geographic_region_greater_canada
				}
				country_is_colonial_or_company = yes
			}
			any_subject_or_below = {
				hidden_trigger = {
					NOT = { c:NEW ?= THIS }
				}
				capital = {
					is_in_geographic_region = geographic_region_greater_canada
				}
				country_is_colonial_or_company = yes
			}
		}
	}

	is_shown_in_lobby = {
		hidden_trigger = {
			NOT = { c:NEW ?= THIS }
		}
		OR = {
			AND = {
				OR = {
					country_has_primary_culture = cu:anglo_canadian
					country_has_primary_culture = cu:franco_canadian
					country_has_primary_culture = cu:acadian
				}
				is_in_geographic_region = geographic_region_greater_canada
			}
			AND = {
				NOT = {
					capital = {
						is_in_geographic_region = geographic_region_greater_canada
					}
				}
				any_subject_or_below = {
					hidden_trigger = {
						NOT = { c:NEW ?= THIS }
					}
					capital = {
						is_in_geographic_region = geographic_region_greater_canada
					}
					count >= 2
				}
			}
		}
	}

	is_shown_when_inactive = {
		any_country_in_greater_canada = {
			hidden_trigger = {
				NOT = { c:NEW ?= THIS }
			}
			is_subject = yes
			country_is_colonial_or_company = yes
			OR = {
				c2c_owns_confederation_states = yes
				overlord = {
					any_subject_or_below = {
						hidden_trigger = {
							NOT = { c:NEW ?= THIS }
						}
						capital = {
							is_in_geographic_region = geographic_region_greater_canada
						}
						count >= 2
					}
				}
			}
		}
	}

	possible = {
		any_country_in_greater_canada = {
			hidden_trigger = {
				NOT = { c:NEW ?= THIS }
			}
			is_subject = yes
			country_is_colonial_or_company = yes
			OR = {
				has_technology_researched = nationalism
				overlord = {
					has_technology_researched = nationalism
				}
			}
		}
	}

	scripted_button = je_canada_unite_can_button

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_country_in_greater_canada = {
				hidden_trigger = {
					NOT = { c:NEW ?= THIS }
				}
				is_subject = yes
				custom_tooltip = {
					text = owns_all_confederate_canada_tt
					c2c_owns_confederation_states = yes
				}
			}
		}
		effect = {
			hidden_effect = {
				s:STATE_ONTARIO ?= {
					random_scope_state = {
						set_hub_name = {
							type = city
							name = HUB_NAME_STATE_ONTARIO_city_ottawa
						}
					}
				}
			}
			je_canada_cleanup = yes
		}
		effect_all_involved = {
			show_as_tooltip = {
				trigger_event = { id = can_aus.5 }
			}
			hidden_effect = {
				if = {
					limit = {
						capital = {
							is_in_geographic_region = geographic_region_greater_canada
						}
						is_subject = yes
						c2c_owns_confederation_states = yes
					}
					trigger_event = { id = can_aus.5 }
					change_tag = CAN
					add_primary_culture = cu:anglo_canadian
					add_primary_culture = cu:franco_canadian
					add_primary_culture = cu:acadian
				}
				else_if = {
					limit = {
						NOT = {
							capital = {
								is_in_geographic_region = geographic_region_greater_canada
							}
						}
					}
					trigger_event = { id = can_aus.6 }
				}
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_country_in_greater_canada = {
					hidden_trigger = {
						NOT = { c:NEW ?= THIS }
					}
					is_subject = yes
					country_is_colonial_or_company = yes
				}
			}
		}
		effect = {
			je_canada_cleanup = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			every_country_in_greater_canada = {
				limit = {
					NOT = { c:NEW ?= THIS }
					has_journal_entry = je_canada
					is_subject = yes
					country_is_colonial_or_company = yes
					NOT = {
						is_involved_in_journal_entry = je_canada
					}
				}
				je:je_canada ?= {
					add_involved_country = prev
				}
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = no
	should_be_pinned_by_default_involved = yes

	can_revolution_inherit = yes
}
