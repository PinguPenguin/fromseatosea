je_c2c_settling_the_prairie = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		c2c_is_canadian_colonial_frontier_country = yes
		any_scope_state = {
			c2c_is_prairie_frontier_state = yes
		}
	}

	possible = {
		c2c_is_canadian_colonial_frontier_country = yes
		any_scope_state = {
			c2c_is_prairie_frontier_state = yes
		}
	}

	scripted_button = c2c_survey_prairie_button
	scripted_button = c2c_settle_prairie_button

	complete = {
		custom_tooltip = {
			text = c2c_settling_the_prairie_complete_tt
			NOT = {
				any_scope_state = {
					c2c_is_prairie_frontier_state = yes
					NOT = {
						c2c_has_settled_prairie_state = yes
					}
				}
			}
		}
	}

	on_complete = {
		add_modifier = {
			name = c2c_prairie_settlement_policy
			days = very_long_modifier_time
		}
		trigger_event = { id = c2c_frontier.5 popup = yes }
	}

	invalid = {
		OR = {
			NOT = { c2c_is_canadian_colonial_frontier_country = yes }
			NOT = {
				any_scope_state = {
					c2c_is_prairie_frontier_state = yes
				}
			}
		}
	}

	weight = 80

	should_be_pinned_by_default_uninvolved_or_context = yes
}
