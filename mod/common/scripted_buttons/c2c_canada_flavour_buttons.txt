je_c2c_responsible_government_button_lobby = {
	name = "je_c2c_responsible_government_button_lobby"
	desc = "je_c2c_responsible_government_button_lobby_desc"

	visible = {
		always = yes
	}

	cooldown = { days = 365 }

	possible = {
		top_overlord = {
			relations:root >= relations_threshold:cordial
		}
		any_scope_character = {
			OR = {
				has_ideology = ideology:ideology_reformer
				has_ideology = ideology:ideology_c2c_reformer_can
			}
		}
	}

	ai_chance = {
		value = 100
	}

	effect = {
		trigger_event = { id = c2c_responsible_gov.2 popup = yes }
	}

	when_taken = {
		set_variable = { name = c2c_lobby_colonial_office_user value = yes days = 180 }
	}
}

je_c2c_responsible_government_button_resolution = {
	name = "je_c2c_responsible_government_button_resolution"
	desc = "je_c2c_responsible_government_button_resolution_desc"

	visible = {
		always = yes
	}

	cooldown = { days = 365 }

	possible = {
		any_interest_group = {
			OR = {
				has_ideology = ideology:ideology_sovereignist
				has_ideology = ideology:ideology_c2c_patriote
				leader = {
					OR = {
						has_ideology = ideology:ideology_radical
						has_ideology = ideology:ideology_reformer
						has_ideology = ideology:ideology_c2c_reformer_can
					}
				}
			}
			ig_clout >= 0.10
		}
	}

	ai_chance = {
		value = 0
	}

	effect = {
		add_liberty_desire = 5
		trigger_event = { id = c2c_responsible_gov.4 popup = yes }
	}

	when_taken = {
		set_variable = { name = c2c_draft_resolutions_user value = yes days = 180 }
	}
}
