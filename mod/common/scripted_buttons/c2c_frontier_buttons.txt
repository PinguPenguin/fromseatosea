c2c_survey_prairie_button = {
	name = c2c_survey_prairie_button
	desc = c2c_survey_prairie_button_desc

	visible = {
		c2c_is_canadian_colonial_frontier_country = yes
		has_journal_entry = je_c2c_settling_the_prairie
	}

	cooldown = { days = 365 }

	possible = {
		c2c_is_canadian_colonial_frontier_country = yes
		OR = {
			has_technology_researched = railways
			has_technology_researched = intensive_agriculture
		}
		any_scope_state = {
			c2c_can_survey_prairie_state = yes
		}
	}

	effect = {
		trigger_event = {
			id = c2c_frontier.1
			popup = yes
		}
	}

	ai_chance = {
		value = 20
		if = {
			limit = {
				OR = {
					is_at_war = yes
					in_default = yes
				}
			}
			add = -100
		}
	}
}
