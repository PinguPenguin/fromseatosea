REPLACE:je_canada_unite_can_button = {
	name = "je_canada_unite_can_button"
	desc = "je_canada_unite_can_button_desc"

	visible = {
		always = yes
	}

	possible = {
		is_at_war = no
		has_technology_researched = nationalism
		NOT = { has_modifier = canada_test_modifier }
		hidden_trigger = {
			NOT = { c:NEW = THIS }
		}
		trigger_if = {
			limit = {
				capital = {
					is_in_geographic_region = geographic_region_greater_canada
				}
			}
			top_overlord ?= {
				relations:root >= relations_threshold:amicable
			}
			custom_tooltip = {
				text = unite_canada_je_can_tt
				any_neighbouring_state = {
					owner = {
						is_ai = yes
						relations:root >= relations_threshold:amicable
						capital = {
							is_in_geographic_region = geographic_region_greater_canada
						}
						is_subject = yes
						NOT = { THIS = ROOT }
						hidden_trigger = {
							NOT = { c:NEW = THIS }
						}
						top_overlord = {
							THIS = root.top_overlord
						}
						this.gdp < root.gdp
					}
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = gbr_can_relations_tt
				any_subject_or_below = {
					capital = {
						is_in_geographic_region = geographic_region_greater_canada
					}
					is_ai = yes
					relations:root >= relations_threshold:amicable
					hidden_trigger = {
						NOT = { c:NEW = THIS }
					}
					count >= 2
				}
			}
		}
	}

	cooldown = { days = short_modifier_time }

	effect = {
		custom_tooltip = unite_canada_tt
		if = {
			limit = {
				capital = {
					is_in_geographic_region = geographic_region_greater_canada
				}
			}
			hidden_effect = {
				trigger_event = { id = can_aus.1 popup = yes }
			}
		}
		else = {
			hidden_effect = {
				trigger_event = { id = can_aus.2 popup = yes }
			}
		}
		add_modifier = {
			name = canada_test_modifier
			days = short_modifier_time
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				has_technology_researched = pan-nationalism
			}
			add = 100
		}
	}
}
