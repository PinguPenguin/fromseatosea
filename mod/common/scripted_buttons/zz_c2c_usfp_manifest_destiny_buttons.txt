REPLACE:usfp_demand_bc_button = {
	name = "usfp_demand_bc_button"
	desc = "usfp_demand_bc_button_desc"

	visible = {
		exists = c:GBR
		NAND = {
			has_claim = s:STATE_BRITISH_COLUMBIA
			has_claim = s:STATE_BC_INTERIOR
			has_claim = s:STATE_UPPER_BC
			has_claim = s:STATE_VANCOUVER_ISLAND
		}
		OR = {
			s:STATE_BRITISH_COLUMBIA = {
				any_scope_state = {
					owner = {
						NOT = { is_country_type = decentralized }
						OR = {
							is_subject_of = c:GBR
							c:GBR = this
						}
					}
				}
			}
			s:STATE_BC_INTERIOR = {
				any_scope_state = {
					owner = {
						NOT = { is_country_type = decentralized }
						OR = {
							is_subject_of = c:GBR
							c:GBR = this
						}
					}
				}
			}
			s:STATE_UPPER_BC = {
				any_scope_state = {
					owner = {
						NOT = { is_country_type = decentralized }
						OR = {
							is_subject_of = c:GBR
							c:GBR = this
						}
					}
				}
			}
			s:STATE_VANCOUVER_ISLAND = {
				any_scope_state = {
					owner = {
						NOT = { is_country_type = decentralized }
						OR = {
							is_subject_of = c:GBR
							c:GBR = this
						}
					}
				}
			}
		}
		has_variable = manifest_destiny_var
	}

	possible = {
		usfp_manifest_destiny_greater_or_equal_trigger = { value = 6 }
		ruler = { usfp_manifest_destiny_pro_expansion_trigger = yes }
	}

	effect = {
		s:STATE_BRITISH_COLUMBIA = { add_claim = root }
		s:STATE_BC_INTERIOR = { add_claim = root }
		s:STATE_UPPER_BC = { add_claim = root }
		s:STATE_VANCOUVER_ISLAND = { add_claim = root }
		change_relations = { country = c:GBR value = -10 }
	}
