# Hub renames from 1836 names to later ones. Hubs can't move or change type after game start, so each
# hub sits where the later town grew and starts under the name of the period post or settlement nearby.
# Runs from on_yearly_pulse (c2c_on_actions.txt); a global variable keeps each rename from repeating.
c2c_update_hub_names = {
	# Port Simpson -> Prince Rupert, when a railway reaches the state
	if = {
		limit = {
			NOT = { has_global_variable = c2c_hub_renamed_upper_bc_port }
			s:STATE_UPPER_BC = {
				any_scope_state = {
					has_building = building_railway
				}
			}
		}
		s:STATE_UPPER_BC = {
			every_scope_state = {
				set_hub_name = {
					type = port
					name = HUB_NAME_STATE_UPPER_BC_port_prince_rupert
				}
			}
		}
		set_global_variable = {
			name = c2c_hub_renamed_upper_bc_port
			value = yes
		}
	}
