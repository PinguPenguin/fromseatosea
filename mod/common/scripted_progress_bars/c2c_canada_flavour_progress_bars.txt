c2c_cement_conservatism_bar = {
	name = "c2c_cement_conservatism_bar_name"
	desc = "c2c_cement_conservatism_bar_desc"
	default = yes

	start_value = 25
	min_value = 0
	max_value = 100

	monthly_progress = {
		if = {
			limit = {
				owner = {
					top_overlord = {
						relations:root >= relations_threshold:amicable
					}
				}
			}
			add = 0.25
		}

		if = {
			limit = {
				owner = {
					any_interest_group = {
						is_in_government = yes
						OR = {
							is_interest_group_type = ig_landowners
							is_interest_group_type = ig_devout
						}
					}
				}
			}
			add = {
				desc = c2c_family_compact_government_progress
				add = ig:ig_landowners.ig_clout
				multiply = 10
			}
		}

		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_religious_schools
				}
				institution:institution_schools.investment >= 2
			}
			add = {
				desc = c2c_family_compact_religious_schools_progress
				institution:institution_schools = {
					add = this.investment
					divide = 3
				}
			}
		}

		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_local_police
				}
				institution:institution_police.investment >= 2
			}
			add = {
				desc = c2c_family_compact_local_police_progress
				institution:institution_police = {
					add = this.investment
					divide = 3
				}
			}
		}

		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_charitable_health_system
				}
				institution:institution_health_system.investment >= 2
			}
			add = {
				desc = c2c_family_compact_charity_hospitals_progress
				institution:institution_health_system = {
					add = this.investment
					divide = 3
				}
			}
		}

		if = {
			limit = {
				owner = {
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_liberal
							is_political_movement_type = movement_radical
						}
						political_movement_support <= 0.05
					}
				}
			}
			add = {
				desc = c2c_family_compact_marginal_reform_progress
				add = 0.5
			}
		}

		if = {
			limit = {
				owner = {
					NOT = { has_variable = c2c_responsible_government_var }
				}
			}
			add = {
				desc = c2c_family_compact_no_responsible_government_progress
				add = 0.25
			}
		}
	}
}
