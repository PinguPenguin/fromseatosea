namespace = c2c_confederation

c2c_confederation.1 = {
	type = country_event
	placement = ROOT

	title = c2c_confederation.1.t
	desc = c2c_confederation.1.d
	flavor = c2c_confederation.1.f

	event_image = {
		video = "southamerica_aristocrats"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		c2c_is_confederation_partner = yes
		global_var:c2c_confederation_interest >= 100
	}

	immediate = {
		c2c_confederation_clear_initiator_flags = yes
		set_variable = {
			name = c2c_confederation_initiator
			days = 365
		}
	}

	option = {
		name = c2c_confederation.1.a
		default_option = yes
		every_country = {
			limit = {
				c2c_is_confederation_partner = yes
				NOT = { THIS = ROOT }
				top_overlord = {
					THIS = root.top_overlord
				}
			}
			trigger_event = {
				id = c2c_confederation.2
				days = 30
				popup = yes
			}
		}
		trigger_event = {
			id = c2c_confederation.3
			days = 90
			popup = yes
		}
	}
}

c2c_confederation.2 = {
	type = country_event
	placement = ROOT

	title = c2c_confederation.2.t
	desc = c2c_confederation.2.d
	flavor = c2c_confederation.2.f

	event_image = {
		video = "southamerica_aristocrats"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		c2c_is_confederation_partner = yes
	}

	immediate = {
		random_country = {
			limit = {
				has_variable = c2c_confederation_initiator
			}
			save_scope_as = c2c_initiator_scope
		}
	}

	option = {
		name = c2c_confederation.2.a
		default_option = yes
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					exists = scope:c2c_initiator_scope
					relations:scope:c2c_initiator_scope >= relations_threshold:amicable
				}
				add = 35
			}
			modifier = {
				trigger = {
					exists = scope:c2c_initiator_scope
					this.gdp < scope:c2c_initiator_scope.gdp
				}
				add = 25
			}
			modifier = {
				trigger = {
					exists = scope:c2c_initiator_scope
					top_overlord = {
						THIS = scope:c2c_initiator_scope.top_overlord
					}
				}
				add = 15
			}
		}
		if = {
			limit = {
				exists = scope:c2c_initiator_scope
			}
			if = {
				limit = { is_player = yes }
				play_as = scope:c2c_initiator_scope
			}
			every_scope_character = {
				limit = { is_historical = yes }
				transfer_character = scope:c2c_initiator_scope
			}
			hidden_effect = {
				every_state_in_canada = {
					limit = {
						has_claim_by = ROOT
					}
					state_region = {
						remove_claim = ROOT
						add_claim = scope:c2c_initiator_scope
					}
				}
			}
			scope:c2c_initiator_scope = {
				annex_with_incorporation = ROOT
			}
		}
	}

	option = {
		name = c2c_confederation.2.b
		ai_chance = {
			base = 75
		}
		if = {
			limit = {
				exists = scope:c2c_initiator_scope
			}
			change_relations = {
				country = scope:c2c_initiator_scope
				value = -10
			}
		}
	}

	after = {
		if = {
			limit = {
				exists = scope:c2c_initiator_scope
			}
			scope:c2c_initiator_scope = {
				trigger_event = {
					id = c2c_confederation.3
					days = 1
					popup = no
				}
			}
		}
	}
}

c2c_confederation.3 = {
	type = country_event
	placement = ROOT

	title = c2c_confederation.3.t
	desc = c2c_confederation.3.d
	flavor = c2c_confederation.3.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	trigger = {
		has_variable = c2c_confederation_initiator
		NOT = { has_variable = c2c_confederation_proclaimed }
	}

	option = {
		name = c2c_confederation.3.a
		default_option = yes
		set_variable = {
			name = c2c_confederation_proclaimed
		}
		add_modifier = {
			name = can_unified_canada
			days = normal_modifier_time
		}
		change_tag = CAN
		add_primary_culture = cu:anglo_canadian
		add_primary_culture = cu:franco_canadian
		add_primary_culture = cu:acadian
		if = {
			limit = {
				is_subject = yes
				NOT = { is_subject_type = subject_type_c2c_self_governing_dominion }
			}
			change_subject_type = subject_type_c2c_self_governing_dominion
		}
		activate_law = law_type:law_parliamentary_republic
		call_election = {
			months = 3
		}
		c2c_confederation_absorb_decentralized_canada_effect = yes
		c2c_confederation_ottawa_hub_effect = yes
	}
}
