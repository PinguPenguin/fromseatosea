namespace = c2c_family_compact

c2c_family_compact.1 = {
	type = country_event
	placement = ROOT

	title = c2c_family_compact.1.t
	desc = c2c_family_compact.1.d
	flavor = c2c_family_compact.1.f

	cooldown = { years = 99 }

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	trigger = {
		c:ONT ?= ROOT
	}

	immediate = {
		if = {
			limit = {
				NOT = { has_modifier = c2c_family_compact_cemented_modifier }
			}
			add_modifier = {
				name = c2c_family_compact_cemented_modifier
				is_decaying = yes
				years = 10
			}
		}
		top_overlord = {
			change_relations = {
				country = ROOT
				value = 15
			}
		}
		if = {
			limit = {
				exists = global_var:c2c_union_act_support
			}
			change_global_variable = {
				name = c2c_union_act_support
				add = 200
			}
		}
		else = {
			set_global_variable = {
				name = c2c_union_act_support
				value = 200
			}
		}
	}

	option = {
		name = c2c_family_compact.1.a
		default_option = yes
	}
}
