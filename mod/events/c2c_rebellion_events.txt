namespace = c2c_rebellion

c2c_rebellion.1 = {
	type = country_event
	placement = ROOT

	title = c2c_rebellion.1.t
	desc = c2c_rebellion.1.d
	flavor = c2c_rebellion.1.f

	cooldown = { years = 99 }

	event_image = {
		video = "ip3_officer_inspecting_peasant_conscripts"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	duration = 3

	trigger = {
		is_subject = yes
	}

	immediate = {
		top_overlord = {
			change_relations = {
				country = ROOT
				value = -15
			}
		}
		random_political_movement = {
			limit = {
				owner = ROOT
				is_political_movement_type = movement_radical
			}
			save_scope_as = radical_movement
		}
		random_interest_group = {
			limit = {
				leader = {
					OR = {
						has_ideology = ideology:ideology_reformer
						has_ideology = ideology:ideology_c2c_reformer_can
					}
				}
			}
			save_scope_as = reformer_ig
		}
	}

	option = {
		name = c2c_rebellion.1.a
		default_option = yes
	}

	option = {
		name = c2c_rebellion.1.b
		add_modifier = {
			name = c2c_rebellion_strong_momentum
			years = 2
		}
	}

	option = {
		name = c2c_rebellion.1.c
		trigger = {
			scope:radical_movement ?= {
				is_being_suppressed = yes
				political_movement_support < 0.15
			}
		}
		scope:radical_movement ?= {
			remove_modifier = c2c_reformer_movement_fury
		}
		trigger_event = {
			id = c2c_rebellion.2
			popup = yes
		}
	}
}

c2c_rebellion.3 = {

	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	trigger = {

	}

	immediate = {
		random_scope_character = {
			limit = {
				has_ideology = ideology:ideology_radical
				popularity > 50
			}
			save_scope_as = popular_radical_char
		}
	}

	option = {
		name = c2c_rebellion.3.a
		default_option = yes
		add_radicals = {
			value = 0.1
			pop_type = farmers
		}
	}

	option = {
		name = c2c_rebellion.3.b
		add_loyalists = {
			pop_type = farmers
			value = 0.05
		}
	}
}
